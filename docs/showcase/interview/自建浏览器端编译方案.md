# 自建浏览器端编译方案

> 本文档详细介绍如何不依赖 Sandpack，自己实现浏览器端的 React/Vue 代码编译和预览。
> 适用于面试深度追问或需要完全控制编译流程的场景。

## 核心原理

### 整体架构


### 关键技术点

| 技术点 | 作用 | 实现方式 |
|--------|------|----------|
| @babel/standalone | 浏览器端编译 TSX/JSX | `transform(code, { presets: ['react', 'typescript'] })` |
| Blob URL | 内联模块代码 | `URL.createObjectURL(new Blob([code], { type: 'application/javascript' }))` |
| Import Maps | 第三方包映射 | `<script type="importmap">{ "imports": { "react": "..." } }</script>` |
| Web Worker | 避免编译阻塞 UI | 将 babel 编译放到 Worker 线程 |
| Babel 插件 | 重写 import 路径 | 自定义 visitor 处理 ImportDeclaration |

## 详细实现

### 1. Babel 编译器

```typescript
// compiler.ts
import { transform, type PluginObj } from '@babel/standalone';

interface CompileOptions {
  filename: string;
  files: Record<string, string>;  // 文件名 → 内容
}

/**
 * 编译单个 TSX/TS 文件
 */
export function compileFile(code: string, options: CompileOptions): string {
  const { filename, files } = options;
  
  const result = transform(code, {
    filename,
    presets: ['react', 'typescript'],
    plugins: [
      // 自定义插件：重写 import 路径
      createImportRewritePlugin(files),
    ],
    retainLines: true,  // 保持行号，便于调试
  });
  
  return result.code || '';
}

/**
 * 创建 import 路径重写插件
 * 
 * 核心逻辑：
 * - 本地模块（./xxx）→ 编译后生成 Blob URL
 * - 第三方包（react）→ 保持原样，由 Import Maps 处理
 */
function createImportRewritePlugin(files: Record<string, string>): PluginObj {
  return {
    visitor: {
      ImportDeclaration(path) {
        const source = path.node.source.value;
        
        // 跳过第三方包，由 Import Maps 处理
        if (!source.startsWith('.') && !source.startsWith('/')) {
          return;
        }
        
        // 解析相对路径
        const resolvedPath = resolveModulePath(source, files);
        if (!resolvedPath) {
          console.warn(`[Compiler] 无法解析模块: ${source}`);
          return;
        }
        
        const fileContent = files[resolvedPath];
        if (!fileContent) {
          console.warn(`[Compiler] 文件不存在: ${resolvedPath}`);
          return;
        }
        
        // 根据文件类型处理
        const ext = resolvedPath.split('.').pop();
        let blobUrl: string;
        
        if (ext === 'css' || ext === 'scss') {
          // CSS 文件：生成注入 <style> 的代码
          blobUrl = createCssBlobUrl(fileContent);
        } else if (ext === 'json') {
          // JSON 文件：导出 JSON 对象
          blobUrl = createJsonBlobUrl(fileContent);
        } else {
          // TS/TSX 文件：递归编译
          const compiled = compileFile(fileContent, { filename: resolvedPath, files });
          blobUrl = createJsBlobUrl(compiled);
        }
        
        // 重写 import 路径为 Blob URL
        path.node.source.value = blobUrl;
      },
    },
  };
}

/**
 * 解析模块路径
 * ./App → ./App.tsx 或 ./App.ts 或 ./App/index.tsx
 */
function resolveModulePath(source: string, files: Record<string, string>): string | null {
  // 移除开头的 ./
  const cleanPath = source.replace(/^\.\//, '');
  
  // 尝试的后缀列表
  const extensions = ['', '.tsx', '.ts', '.jsx', '.js', '/index.tsx', '/index.ts'];
  
  for (const ext of extensions) {
    const fullPath = cleanPath + ext;
    if (files[fullPath] || files[`src/${fullPath}`]) {
      return files[fullPath] ? fullPath : `src/${fullPath}`;
    }
  }
  
  return null;
}
```

### 2. Blob URL 生成器

```typescript
// blobUrlGenerator.ts

/**
 * 创建 JavaScript Blob URL
 */
export function createJsBlobUrl(code: string): string {
  const blob = new Blob([code], { type: 'application/javascript' });
  return URL.createObjectURL(blob);
}

/**
 * 创建 CSS Blob URL
 * CSS 需要包装成 JS 模块，动态注入 <style>
 */
export function createCssBlobUrl(cssContent: string): string {
  // 转义 CSS 中的特殊字符
  const escaped = cssContent
    .replace(/\\/g, '\\\\')
    .replace(/`/g, '\\`')
    .replace(/\$/g, '\\$');
  
  const jsCode = `
    const style = document.createElement('style');
    style.textContent = \`${escaped}\`;
    document.head.appendChild(style);
  `;
  
  return createJsBlobUrl(jsCode);
}

/**
 * 创建 JSON Blob URL
 * JSON 需要包装成 ES Module 导出
 */
export function createJsonBlobUrl(jsonContent: string): string {
  const jsCode = `export default ${jsonContent};`;
  return createJsBlobUrl(jsCode);
}

/**
 * 清理 Blob URL（防止内存泄漏）
 */
export function revokeBlobUrls(urls: string[]): void {
  urls.forEach(url => {
    if (url.startsWith('blob:')) {
      URL.revokeObjectURL(url);
    }
  });
}
```

### 3. Import Maps 生成器

```typescript
// importMapGenerator.ts

/**
 * 第三方包 CDN 映射
 * 使用 esm.sh 提供 ES Module 格式的包
 */
const CDN_BASE = 'https://esm.sh';

/**
 * 常用包的版本锁定
 */
const PACKAGE_VERSIONS: Record<string, string> = {
  'react': '18.2.0',
  'react-dom': '18.2.0',
  'antd': '5.12.0',
  'react-router-dom': '6.20.0',
  '@ant-design/icons': '5.2.6',
};

/**
 * 生成 Import Maps
 */
export function generateImportMap(dependencies: Record<string, string>): string {
  const imports: Record<string, string> = {};
  
  for (const [pkg, version] of Object.entries(dependencies)) {
    // 使用锁定版本或 package.json 中的版本
    const resolvedVersion = PACKAGE_VERSIONS[pkg] || version.replace(/^\^|~/, '');
    imports[pkg] = `${CDN_BASE}/${pkg}@${resolvedVersion}`;
    
    // 处理子路径导入（如 react-dom/client）
    imports[`${pkg}/`] = `${CDN_BASE}/${pkg}@${resolvedVersion}/`;
  }
  
  // 添加 React JSX Runtime（babel 编译后需要）
  imports['react/jsx-runtime'] = `${CDN_BASE}/react@${PACKAGE_VERSIONS['react']}/jsx-runtime`;
  
  return JSON.stringify({ imports }, null, 2);
}

/**
 * 示例输出：
 * {
 *   "imports": {
 *     "react": "https://esm.sh/react@18.2.0",
 *     "react/": "https://esm.sh/react@18.2.0/",
 *     "react-dom": "https://esm.sh/react-dom@18.2.0",
 *     "react-dom/": "https://esm.sh/react-dom@18.2.0/",
 *     "react/jsx-runtime": "https://esm.sh/react@18.2.0/jsx-runtime"
 *   }
 * }
 */
```

### 5. Web Worker 编译器

```typescript
// compiler.worker.ts
import { transform } from '@babel/standalone';

// Worker 消息类型
interface CompileRequest {
  type: 'COMPILE';
  files: Record<string, string>;
  entryFile: string;
}

interface CompileResponse {
  type: 'COMPILED';
  html: string;
  error?: string;
}

self.onmessage = async (event: MessageEvent<CompileRequest>) => {
  const { type, files, entryFile } = event.data;
  
  if (type !== 'COMPILE') return;
  
  try {
    // 1. 编译入口文件（递归编译所有依赖）
    const compiledEntry = compileFile(files[entryFile], {
      filename: entryFile,
      files,
    });
    
    // 2. 生成入口 Blob URL
    const entryBlobUrl = createJsBlobUrl(compiledEntry);
    
    // 3. 从 package.json 提取依赖
    const packageJson = JSON.parse(files['package.json'] || '{}');
    const dependencies = packageJson.dependencies || {};
    
    // 4. 生成 Import Maps
    const importMap = generateImportMap(dependencies);
    
    // 5. 生成入口代码
    const entryCode = generateEntryCode(entryBlobUrl);
    
    // 6. 生成完整 HTML
    const html = generatePreviewHtml({ importMap, entryCode });
    
    self.postMessage({ type: 'COMPILED', html } as CompileResponse);
    
  } catch (error) {
    self.postMessage({
      type: 'COMPILED',
      html: '',
      error: error instanceof Error ? error.message : String(error),
    } as CompileResponse);
  }
};
```

### 6. 预览组件

```tsx
// CodePreview.tsx
import { useEffect, useRef, useState } from 'react';
import type { IGeneratedFile } from '@lowcode/schema';
import CompilerWorker from './compiler.worker?worker';

interface CodePreviewProps {
  files: IGeneratedFile[];
}

export function CodePreview({ files }: CodePreviewProps) {
  const iframeRef = useRef<HTMLIFrameElement>(null);
  const workerRef = useRef<Worker>();
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState<string>();
  
  useEffect(() => {
    // 初始化 Worker
    if (!workerRef.current) {
      workerRef.current = new CompilerWorker();
      workerRef.current.onmessage = (event) => {
        const { type, html, error } = event.data;
        if (type === 'COMPILED') {
          setLoading(false);
          if (error) {
            setError(error);
          } else {
            // 设置 iframe src 为 Blob URL
            const blob = new Blob([html], { type: 'text/html' });
            const url = URL.createObjectURL(blob);
            if (iframeRef.current) {
              iframeRef.current.src = url;
            }
          }
        }
      };
    }
    
    // 转换文件格式
    const fileMap: Record<string, string> = {};
    files.forEach(f => {
      fileMap[f.filePath] = f.content;
    });
    
    // 发送编译请求
    setLoading(true);
    setError(undefined);
    workerRef.current.postMessage({
      type: 'COMPILE',
      files: fileMap,
      entryFile: 'src/main.tsx',
    });
    
    return () => {
      workerRef.current?.terminate();
    };
  }, [files]);
  
  if (error) {
    return (
      <div style={{ padding: 20, color: '#ff4d4f' }}>
        <h3>编译错误</h3>
        <pre>{error}</pre>
      </div>
    );
  }
  
  return (
    <div style={{ position: 'relative', width: '100%', height: '100%' }}>
      {loading && (
        <div style={{
          position: 'absolute',
          inset: 0,
          display: 'flex',
          alignItems: 'center',
          justifyContent: 'center',
          background: '#f5f5f5',
        }}>
          编译中...
        </div>
      )}
      <iframe
        ref={iframeRef}
        sandbox="allow-scripts allow-same-origin"
        style={{
          width: '100%',
          height: '100%',
          border: 'none',
          opacity: loading ? 0 : 1,
        }}
      />
    </div>
  );
}
```

## CSS Module 处理

CSS Module 是最复杂的部分，需要在编译时生成类名映射。

### 方案 A：简化处理（推荐）

出码时不使用 CSS Module，改为内联样式或普通 CSS：

```typescript

// 或普通 CSS 类名
<div className="my-component">Hello</div>
```

### 方案 B：模拟 CSS Module

```typescript
// cssModuleCompiler.ts

/**
 * 编译 CSS Module
 * 输入：.module.css 文件内容
 * 输出：{ css: string, classMap: Record<string, string> }
 */
export function compileCssModule(css: string, filename: string): {
  css: string;
  classMap: Record<string, string>;
} {
  const classMap: Record<string, string> = {};
  const hash = simpleHash(filename);
  
  // 替换类名，添加 hash 后缀
  const compiledCss = css.replace(/\.([a-zA-Z_][a-zA-Z0-9_-]*)/g, (match, className) => {
    const hashedName = `${className}_${hash}`;
    classMap[className] = hashedName;
    return `.${hashedName}`;
  });
  
  return { css: compiledCss, classMap };
}

/**
 * 生成 CSS Module 的 JS 模块
 */
export function createCssModuleBlobUrl(css: string, filename: string): string {
  const { css: compiledCss, classMap } = compileCssModule(css, filename);
  
  const jsCode = `
    // 注入样式
    const style = document.createElement('style');
    style.textContent = ${JSON.stringify(compiledCss)};
    document.head.appendChild(style);
    
    // 导出类名映射
    export default ${JSON.stringify(classMap)};
  `;
  
  return createJsBlobUrl(jsCode);
}

function simpleHash(str: string): string {
  let hash = 0;
  for (let i = 0; i < str.length; i++) {
    hash = ((hash << 5) - hash) + str.charCodeAt(i);
    hash |= 0;
  }
  return Math.abs(hash).toString(36).slice(0, 5);
}
```

## Vue 支持

Vue 需要额外处理 SFC（单文件组件）。

### 方案 A：使用 @vue/compiler-sfc

```typescript
import { parse, compileScript, compileTemplate, compileStyle } from '@vue/compiler-sfc';

export function compileVueSfc(code: string, filename: string) {
  const { descriptor } = parse(code, { filename });
  
  // 编译 <script>
  const script = compileScript(descriptor, { id: filename });
  
  // 编译 <template>
  const template = compileTemplate({
    source: descriptor.template?.content || '',
    filename,
    id: filename,
  });
  
  // 编译 <style>
  const styles = descriptor.styles.map(style => 
    compileStyle({
      source: style.content,
      filename,
      id: filename,
      scoped: style.scoped,
    })
  );
  
  // 组装最终代码
  return `
    ${script.content}
    ${template.code}
    ${styles.map(s => `
      const style = document.createElement('style');
      style.textContent = ${JSON.stringify(s.code)};
      document.head.appendChild(style);
    `).join('\n')}
  `;
}
```

## 性能优化

### 1. 编译缓存

```typescript
const compileCache = new Map<string, string>();

function compileWithCache(code: string, filename: string): string {
  const cacheKey = `${filename}:${simpleHash(code)}`;
  
  if (compileCache.has(cacheKey)) {
    return compileCache.get(cacheKey)!;
  }
  
  const result = compileFile(code, { filename, files: {} });
  compileCache.set(cacheKey, result);
  
  return result;
}
```

### 2. 增量编译

只重新编译变更的文件：

```typescript
function incrementalCompile(
  prevFiles: Record<string, string>,
  nextFiles: Record<string, string>,
  prevBlobUrls: Record<string, string>,
): Record<string, string> {
  const newBlobUrls = { ...prevBlobUrls };
  
  for (const [path, content] of Object.entries(nextFiles)) {
    if (prevFiles[path] !== content) {
      // 文件变更，重新编译
      const compiled = compileFile(content, { filename: path, files: nextFiles });
      
      // 释放旧的 Blob URL
      if (newBlobUrls[path]) {
        URL.revokeObjectURL(newBlobUrls[path]);
      }
      
      newBlobUrls[path] = createJsBlobUrl(compiled);
    }
  }
  
  return newBlobUrls;
}
```

## 错误处理

### 编译时错误

```typescript
try {
  const result = transform(code, options);
} catch (error) {
  if (error instanceof SyntaxError) {
    // 语法错误，显示行号
    const match = error.message.match(/\((\d+):(\d+)\)/);
    if (match) {
      const [, line, column] = match;
      return {
        error: `语法错误 at ${line}:${column}\n${error.message}`,
      };
    }
  }
  throw error;
}
```

### 运行时错误

通过 iframe 的 `window.onerror` 捕获，并通过 `postMessage` 传递给父窗口：

```typescript
// iframe 内
window.onerror = (message, source, line, col, error) => {
  parent.postMessage({
    type: 'RUNTIME_ERROR',
    error: { message, source, line, col },
  }, '*');
};

// 父窗口
window.addEventListener('message', (event) => {
  if (event.data.type === 'RUNTIME_ERROR') {
    setError(event.data.error);
  }
});
```

> 1. **浏览器端编译**：用 `@babel/standalone` 编译 TSX/TS 代码，它是 Babel 的浏览器版本，支持 React 和 TypeScript preset。
>
> 2. **模块解析**：写了一个 Babel 插件处理 import 语句。本地模块编译后生成 Blob URL 内联，第三方包通过 Import Maps 映射到 esm.sh CDN。
>
> 3. **CSS 处理**：CSS 文件包装成 JS 模块，动态创建 `<style>` 标签注入。CSS Module 需要额外处理类名 hash。
>
> 4. **性能优化**：编译逻辑放到 Web Worker 避免阻塞 UI，加上编译缓存和防抖。
>
> 5. **沙盒隔离**：最终生成的 HTML 通过 Blob URL 设置为 iframe 的 src，用 sandbox 属性隔离。
>
> 这个方案的技术挑战主要在模块解析和 CSS Module 处理，但可以完全控制编译流程，也是很好的学习 Babel 和浏览器模块系统的机会。"
